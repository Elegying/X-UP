cmake_minimum_required(VERSION 3.22)
project(xup_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/llama.cpp EXCLUDE_FROM_ALL)
target_compile_definitions(ggml-base PRIVATE XUP_LEGACY_STQ_MODEL)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
 add_library(xup_stq_fast OBJECT stq-dotprod.c cpu-capabilities.c)
 target_include_directories(xup_stq_fast PRIVATE vendor/llama.cpp/ggml/include vendor/llama.cpp/ggml/src vendor/llama.cpp/ggml/src/ggml-cpu vendor/llama.cpp/ggml/src/ggml-cpu/arch/arm)
 set_source_files_properties(stq-dotprod.c PROPERTIES COMPILE_OPTIONS "-march=armv8.2-a+dotprod")
 target_sources(ggml-cpu PRIVATE $<TARGET_OBJECTS:xup_stq_fast>)
 target_compile_definitions(ggml-cpu PRIVATE XUP_STQ_RUNTIME_DOTPROD)
endif()

add_library(xup_engine STATIC engine.cpp)
target_link_libraries(xup_engine PUBLIC llama)
target_include_directories(xup_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(ANDROID)
 add_library(xup_translation SHARED jni.cpp)
 target_link_libraries(xup_translation PRIVATE xup_engine log)
 target_link_options(xup_translation PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_executable(xup_kernel_test kernel_test.c)
 target_include_directories(xup_kernel_test PRIVATE vendor/llama.cpp/ggml/src)
 target_link_libraries(xup_kernel_test PRIVATE ggml-cpu ggml-base)
 add_executable(xup_model_test host_test.cpp)
 target_link_libraries(xup_model_test PRIVATE xup_engine)
endif()
